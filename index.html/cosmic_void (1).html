async function send(){

  const inp = document.getElementById('cin');

  const txt = inp.value.trim();

  if(!txt) return;

  addM('me', txt);

  inp.value = '';

  showTyp();


  try {

    // 第一步：发起对话

    const res = await fetch('https://api.coze.cn/v3/chat', {

      method: 'POST',

      headers: {

        'Authorization': 'Bearer 你的API_TOKEN放这里',

        'Content-Type': 'application/json'

      },

      body: JSON.stringify({

        bot_id: '你的BOT_ID放这里',

        user_id: 'user_001',

        stream: false,

        auto_save_history: true,

        additional_messages: [{

          role: 'user',

          content: txt,

          content_type: 'text'

        }]

      })

    });

    const data = await res.json();

    const chatId = data.data.id;

    const convId = data.data.conversation_id;


    // 第二步：轮询等待回复（最多等10秒）

    let reply = '正在感应宇宙频率…';

    for(let i = 0; i < 20; i++){

      await new Promise(r => setTimeout(r, 500));

      const r2 = await fetch(

        `https://api.coze.cn/v3/chat/retrieve?chat_id=${chatId}&conversation_id=${convId}`,

        { headers: { 'Authorization': 'Bearer 你的API_TOKEN放这里' }}

      );

      const d2 = await r2.json();

      if(d2.data.status === 'completed'){

        // 第三步：获取消息内容

        const r3 = await fetch(

          `https://api.coze.cn/v3/chat/message/list?chat_id=${chatId}&conversation_id=${convId}`,

          { headers: { 'Authorization': 'Bearer 你的API_TOKEN放这里' }}

        );

        const d3 = await r3.json();

        const aiMsg = d3.data.find(m => m.role === 'assistant' && m.type === 'answer');

        if(aiMsg) reply = aiMsg.content;

        break;

      }

    }

    const t = document.getElementById('typ');

    if(t) t.remove();

    addM('ai', reply);


  } catch(e) {

    const t = document.getElementById('typ');

    if(t) t.remove();

    addM('ai', '链接暂时中断，请稍后再试。');

  }

}8D 显化
